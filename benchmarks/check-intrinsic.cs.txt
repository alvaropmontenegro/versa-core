using System.Reflection;
using System.Text.Json.Nodes;
var assembly = Assembly.Load("versa-core");
var extractor = assembly.GetType("VersaCore.SnapshotExtractor")!;
var pipeline = assembly.GetType("VersaCore.IntrinsicAuditPipeline")!;
int checks = 0;
void Check(bool ok, string name) { if (!ok) throw new Exception(name); checks++; }
async Task<string> Outline(string html) {
    var parse = (Task)extractor.GetMethod("ParseDocumentAsync")!.Invoke(null, new object[]{html})!;
    await parse;
    var document = parse.GetType().GetProperty("Result")!.GetValue(parse);
    var build = (Task)extractor.GetMethod("BuildStructuralOutlineAsync")!.Invoke(null, new[]{document})!;
    await build;
    var result = build.GetType().GetProperty("Result")!.GetValue(build)!;
    return (string)result.GetType().GetProperty("Markdown")!.GetValue(result)!;
}
Check(await Outline("<div><strong>Written By:</strong> Lindsay Sullivan</div>") == "<ui_text>Written By: Lindsay Sullivan</ui_text>", "author duplicate");
Check(await Outline("<div><span>Hello </span><em>world</em><span>.</span></div>") == "<ui_text>Hello world.</ui_text>", "inline-only wrapper");
Check(await Outline("<div>Before<p>Middle</p>After</div>") == "<ui_text>Before</ui_text>\n\n<prose>Middle</prose>\n\n<ui_text>After</ui_text>", "reading order");
Check(await Outline("<p>Talk <a href='/'>with us</a>.</p>") == "<prose>Talk with us.</prose>", "inline punctuation");
Check(await Outline("<h1>Our <em>title</em></h1>") == "<heading level=\"1\">Our title</heading>", "inline heading");
Check(await Outline("<div>A<br>B</div>") == "<ui_text>A B</ui_text>", "line break");
Check(await Outline("<div>A<span hidden>hidden</span> B</div>") == "<ui_text>A B</ui_text>", "hidden child");
Check(await Outline("<style>.number{display:block}</style><div><span class='number'>Focused</span><span>Less tool sprawl</span></div>") == "<ui_text>Focused</ui_text>\n\n<ui_text>Less tool sprawl</ui_text>", "CSS component boundaries preserved");
Check(await Outline("<style>.hidden{display:none}</style><div>Visible<span class='hidden'> noise</span></div>") == "<ui_text>Visible</ui_text>", "computed display none excluded");
var validate = pipeline.GetMethod("HasValidCorrection", BindingFlags.NonPublic|BindingFlags.Static)!;
JsonObject Evidence(string quote, string span, string? correction, string status="safe") => new() { ["defectConfirmed"]=true, ["evidenceBasis"]="mandatory_linguistic_rule", ["originalHasValidReading"]=false, ["originalReading"]="", ["quote"]=quote, ["problemSpan"]=span, ["correctedQuote"]=correction, ["correctionStatus"]=status };
bool Valid(JsonObject e) => (bool)validate.Invoke(null,new object[]{e})!;
var safe = Evidence("The developmer works.","developmer","The developer works.");
Check(Valid(safe) && safe["edit"]!["start"]!.GetValue<int>() == 4, "localized edit");
Check(!Valid(Evidence("The developmer works.","developmer","A developer worked.")), "outside changes rejected");
Check(Valid(Evidence("They delivers and lend support.","delivers and lend",null,"requires_review")), "review without correction");
Check(!Valid(Evidence("They delivers and lend support.","delivers and lend","They deliver and lend support.","requires_review")), "review cannot substitute");
Check(!Valid(Evidence("The developmer works.","developmer",null)), "safe needs correction");
Check(!Valid(Evidence("in in", "in", "in")), "ambiguous span rejected");
Check(Valid(Evidence("We work in in teams.", "in in", "We work in teams.")), "repeated token repair");
Check(!Valid(Evidence("Text", "Text", "Text")), "no-op rejected");
string Schema(string name) => (string)pipeline.GetField(name, BindingFlags.NonPublic|BindingFlags.Static)!.GetRawConstantValue()!;
var editorialSchema = JsonNode.Parse(Schema("EditorialInspectionSchema"))!;
var writingSchema = JsonNode.Parse(Schema("WritingInspectionSchema"))!;
Check(editorialSchema["properties"]!["writingErrors"] is null, "editorial contract excludes writing errors");
Check(writingSchema["properties"]!.AsObject().Count == 1 && writingSchema["properties"]!["writingErrors"] is not null, "writing contract has only writing errors");
Check(editorialSchema["properties"]!.AsObject().Count == 2, "editorial contract retains its two categories");
var correctionPolicy = pipeline.GetMethod("ApplyWritingCorrectionPolicy", BindingFlags.NonPublic|BindingFlags.Static)!;
var agreement = Evidence("A and B delivers and lend.", "lend", "A and B delivers and lends.");
agreement["errorType"] = "agreement";
correctionPolicy.Invoke(null,new object[]{agreement});
Check(agreement["correctedQuote"] is null && Valid(agreement), "agreement still requires review without verifier");
correctionPolicy.Invoke(null,new object[]{safe});
Check(safe["correctedQuote"]!.GetValue<string>() == "The developer works.", "non-agreement correction preserved");
var mislabeled = Evidence("Our team attend events.", "attend", "Our team attends events.");
mislabeled["errorType"] = "grammar";
correctionPolicy.Invoke(null,new object[]{mislabeled});
Check(mislabeled["correctedQuote"] is null, "grammar label cannot bypass agreement policy");
var duplicate = Evidence("We work in in teams.", "in in", "We work in teams.");
duplicate["errorType"] = "grammar";
correctionPolicy.Invoke(null,new object[]{duplicate});
Check(duplicate["correctedQuote"] is not null, "mechanically localized duplicate correction retained");
var conversion = pipeline.GetMethod("ConvertWritingFindings", BindingFlags.NonPublic|BindingFlags.Static)!;
var editorialConversion = pipeline.GetMethod("ConvertEditorialFindings", BindingFlags.NonPublic|BindingFlags.Static)!;
var payload = new JsonObject { ["writingErrors"] = new JsonArray(safe.DeepClone()) };
Check(((JsonArray)conversion.Invoke(null,new object[]{payload})!).Count == 1, "writing stage creates writing finding");
var uncertain = safe.DeepClone().AsObject();
uncertain["defectConfirmed"] = false;
Check(((JsonArray)conversion.Invoke(null,new object[]{new JsonObject { ["writingErrors"] = new JsonArray(uncertain) }})!).Count == 0, "uncertain defect cannot create issue");
JsonArray Convert(JsonObject e) => (JsonArray)conversion.Invoke(null,new object[]{new JsonObject { ["writingErrors"] = new JsonArray(e.DeepClone()) }})!;
var alternative = safe.DeepClone().AsObject();
alternative["originalHasValidReading"] = true;
alternative["originalReading"] = "An accepted regional reading makes the original valid.";
Check(Convert(alternative).Count == 0, "accepted reading overrides affirmative model verdict");
var style = safe.DeepClone().AsObject(); style["evidenceBasis"]="editorial_convention";
Check(Convert(style).Count == 0, "editorial convention cannot create defect");
style["evidenceBasis"]="name_consistency";
Check(Convert(style).Count == 0, "name variation cannot authorize correction");
var realAgreement = Evidence("Each author write clearly.", "write", null, "requires_review");
realAgreement["errorType"]="agreement";
realAgreement["agreementAnalysis"]=new JsonObject { ["subjectQuote"]="Each author", ["verbQuote"]="write", ["subjectReading"]="singular" };
Check(Convert(realAgreement).Count == 1, "demonstrated agreement defect retained");
realAgreement["agreementAnalysis"]!["subjectReading"]="ambiguous";
Check(Convert(realAgreement).Count == 0, "ambiguous attachment cannot establish agreement defect");
realAgreement["errorType"]="grammar";
Check(Convert(realAgreement).Count == 0, "grammar label cannot override ambiguous agreement evidence");
var candidatesMethod = pipeline.GetMethod("ConvertWritingCandidates", BindingFlags.NonPublic|BindingFlags.Static)!;
var candidateRoot = new JsonObject { ["writingErrors"] = new JsonArray(style.DeepClone()) };
var candidates = (JsonArray)candidatesMethod.Invoke(null,new object[]{candidateRoot, "<prose>The developmer works.</prose>"})!;
Check(candidates.Count == 1 && candidates[0]!["correctedQuote"] is null && !candidates[0]!["affectsClassification"]!.GetValue<bool>() && candidates[0]!["evidenceStatus"]!.GetValue<string>() == "literal_quote_valid", "unconfirmed candidate preserved without edit or classification effect");
var caseOnly = Evidence("Available Worldwide", "Worldwide", "Available worldwide");
caseOnly["blockRole"] = "ui_text";
Check((bool)pipeline.GetMethod("IsCaseOnlyStyleRevision", BindingFlags.NonPublic|BindingFlags.Static)!.Invoke(null,new object[]{caseOnly})!, "UI capitalization preference rejected");
Check(((JsonArray)editorialConversion.Invoke(null,new object[]{payload})!).Count == 0, "editorial stage cannot create writing finding");
var filler = new JsonObject { ["publishedPlaceholders"] = new JsonArray(new JsonObject { ["evidenceQuote"] = "Lorem ipsum", ["reason"] = "Published filler" }) };
Check(((JsonArray)conversion.Invoke(null,new object[]{filler})!).Count == 0 && ((JsonArray)editorialConversion.Invoke(null,new object[]{filler})!).Count == 1, "placeholder belongs only to editorial stage");
Console.WriteLine($"{checks} local checks passed.");
